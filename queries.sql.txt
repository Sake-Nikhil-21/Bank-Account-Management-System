USE Bank_Account_Management_System;

-- 1. Display all customers
SELECT * FROM Customer;

-- 2. Display all bank accounts
SELECT * FROM Account;

-- 3. Display customer names and account numbers
SELECT Customer.Name, Account.Account_No, Account.Account_Type, Account.Balance
FROM Customer JOIN Account
ON Customer.Customer_ID = Account.Customer_ID;

-- 4. Display transaction history
SELECT * FROM Transaction_Table ORDER BY Transaction_Date;

-- 5. Accounts with balance greater than 20000
SELECT Account_No, Account_Type, Balance
FROM Account WHERE Balance > 20000;

-- 6. Total money in all accounts
SELECT SUM(Balance) AS Total_Bank_Balance FROM Account;

-- 7. Customers from Hyderabad
SELECT * FROM Customer WHERE Address = 'Hyderabad';

-- 8. Highest account balance
SELECT MAX(Balance) AS Highest_Balance FROM Account;
